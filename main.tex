\subsection{Partial order}
\begin{frame}{Partial order}
\uncover<1->{Define the partial order $(\Rm,\preceq)$, for $\vu, \vv\in\Rm$, as
\begin{align*}
&&\vu&\pleq \vv&& \\\Leftrightarrow&& u_i&\leq v_i&&\forall i\in\range{m}
\end{align*}}

\uncover<2->{Examples:
\begin{itemize}
\item<2-> $\begin{bmatrix}2\\1\end{bmatrix} \preceq \begin{bmatrix}2\\2\end{bmatrix}$
\item<3-> $\begin{bmatrix}2\\1\end{bmatrix}$ and $\begin{bmatrix}1\\2\end{bmatrix}$ are not comparable with this partial order.
\end{itemize}}
\end{frame}

\subsection{Pareto optimality}
\begin{frame}{Pareto optimality}

\subsection{Jacobian descent}
\begin{frame}{Jacobian descent}
Goal: minimize $\vf : \Rn \to \Rm$ according to the $\preceq$ partial order.
\pause
For any $\vx\in\Rn$, the Jacobian of $\vf$ at $\vx$, is
\begin{align*}
  \jac \vf(\vx)=
  \begin{bmatrix}
    \nabla f_{\textcolor[HTML]{FF0000}{1}}(\vx)^\T\\
    \nabla f_{\textcolor[HTML]{FF8000}{2}}(\vx)^\T\\
    \vdots\\
    \nabla f_{\textcolor[HTML]{DDAA00}{m}}(\vx)^\T
  \end{bmatrix}
  =
  \begin{bmatrix}
    \frac{\partial}{\partial x_{\textcolor[HTML]{0000FF}{1}}} f_{\textcolor[HTML]{FF0000}{1}}(\vx)&\frac{\partial}{\partial x_{\textcolor[HTML]{00AACC}{2}}} f_{\textcolor[HTML]{FF0000}{1}}(\vx)&\cdots&\frac{\partial}{\partial x_{\textcolor[HTML]{00AA00}{n}}} f_{\textcolor[HTML]{FF0000}{1}}(\vx)\\
    \frac{\partial}{\partial x_{\textcolor[HTML]{0000FF}{1}}} f_{\textcolor[HTML]{FF8000}{2}}(\vx)&\frac{\partial}{\partial x_{\textcolor[HTML]{00AACC}{2}}} f_{\textcolor[HTML]{FF8000}{2}}(\vx)&\cdots&\frac{\partial}{\partial x_{\textcolor[HTML]{00AA00}{n}}} f_{\textcolor[HTML]{FF8000}{2}}(\vx)\\
    \vdots&\vdots&\ddots&\vdots\\
    \frac{\partial}{\partial x_{\textcolor[HTML]{0000FF}{1}}} f_{\textcolor[HTML]{DDAA00}{m}}(\vx)&\frac{\partial}{\partial x_{\textcolor[HTML]{00AACC}{2}}} f_{\textcolor[HTML]{DDAA00}{m}}(\vx)&\cdots&\frac{\partial}{\partial x_{\textcolor[HTML]{00AA00}{n}}} f_{\textcolor[HTML]{DDAA00}{m}}(\vx)\\
  \end{bmatrix}
\end{align*}

\pause
For a small enough update $\vy\in\Rn$, Taylor's theorem states
\begin{align*}
\vf(\vx+\vy)\approx\vf(\vx)+ \jac \vf(\vx)\cdot \vy \visible<4->{\textcolor{gray}{{}\preceq \vf(\vx)}}
\end{align*}

\pause
\pause
We select $\vy=-\eta\A\big(\jac \vf(\vx)\big)\in\Rn$.

\end{frame}

\begin{frame}[t]{Jacobian descent}
\vspace{1cm}
The mapping $\A:\Rmn\to\Rn$ is called an {\bf aggregator}, it parameterizes {\bf Jacobian descent}:

\vspace{0.5cm}
\only<2->{
\begin{algorithm}[H]
\caption{Jacobian descent with aggregator $\A$}
\DontPrintSemicolon
\KwIn{$\vx\in\Rn$, $0<\eta$, $T\in\mathbb N$, $\A:\Rmn\to\Rn$}
\For{$t \gets 1$ \KwTo $T$}{
    $\vx\gets \vx - \eta \A\big(\jac \vf(\vx)\big)$\;
}
\KwOut{$\vx$}
\end{algorithm}}
\end{frame}

\subsection{Unconflicting projection of gradients (UPGrad)}
\begin{frame}{Formal definition of UPGrad}
\begin{columns}[T]
\column{0.5\textwidth}
Given $J\in\Rmn$, let ${\color{color1}C_J}$ be the cone $\{ \vy\in\Rn : \0\pleq J\vy \}$.

\begin{flalign*}
\textcolor{darkgray}{p_i}=\argmin_{\vy\in {\color{color1}C_J}}\| \nabla f_i-\vy\|^2&&
\end{flalign*}
\begin{flalign*}
\textcolor{darkpurple}{\upgrad(J)}=\frac{1}{m}\sum_{i\in\range{m}} \textcolor{darkgray}{p_i}&&
\end{flalign*}

\end{columns}
\end{frame}
