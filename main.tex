\subsection{Multi-objective optimization}
\begin{frame}{Multi-objective optimization}
\uncover<1->{Define the partial order $(\Rm,\preceq)$, for $\vu, \vv\in\Rm$, as
\begin{align*}
&&\vu&\pleq \vv&& \\\Leftrightarrow&& u_i&\leq v_i&&\forall i\in\range{m}
\end{align*}}

\uncover<2->{Examples:
\begin{itemize}
\item<2-> $\begin{bmatrix}2\\1\end{bmatrix} \preceq \begin{bmatrix}2\\2\end{bmatrix}$
\item<3-> $\begin{bmatrix}2\\1\end{bmatrix}$ and $\begin{bmatrix}1\\2\end{bmatrix}$ are not comparable with this partial order.
\end{itemize}}
\end{frame}

\subsection{First-order Taylor approximation}
\begin{frame}{First-order Taylor approximation}
\uncover<1->{
For any $\vx\in\Rn$, the Jacobian of $\vf$ at $\vx$, is
\begin{align*}
  \jac \vf(\vx)= 
  \begin{bmatrix}
    \textcolor[HTML]{FF0000}{\nabla f_1(\vx)^\T}\\
    \textcolor[HTML]{0000FF}{\nabla f_2(\vx)^\T}\\
    \vdots\\
    \textcolor[HTML]{008000}{\nabla f_m(\vx)^\T}
  \end{bmatrix}
  = 
  \begin{bmatrix}
    \textcolor[HTML]{FF0000}{\frac{\partial}{\partial x_1} f_1(\vx)}&\textcolor[HTML]{FF0000}{\frac{\partial}{\partial x_2} f_1(\vx)}&\textcolor[HTML]{FF0000}{\cdots}&\textcolor[HTML]{FF0000}{\frac{\partial}{\partial x_n} f_1(\vx)}\\
    \textcolor[HTML]{0000FF}{\frac{\partial}{\partial x_1} f_2(\vx)}&\textcolor[HTML]{0000FF}{\frac{\partial}{\partial x_2} f_2(\vx)}&\textcolor[HTML]{0000FF}{\cdots}&\textcolor[HTML]{0000FF}{\frac{\partial}{\partial x_n} f_2(\vx)}\\
    \vdots&\vdots&\ddots&\vdots\\
    \textcolor[HTML]{008000}{\frac{\partial}{\partial x_1} f_m(\vx)}&\textcolor[HTML]{008000}{\frac{\partial}{\partial x_2} f_m(\vx)}&\textcolor[HTML]{008000}{\cdots}&\textcolor[HTML]{008000}{\frac{\partial}{\partial x_n} f_m(\vx)}\\
  \end{bmatrix}
\end{align*}
}

\uncover<2->{
For a small enough update $\vy\in\Rn$, Taylor's theorem states
\begin{align*}
\vf(\vx+\vy)\approx\vf(\vx)+ \jac \vf(\vx)\cdot \vy
\end{align*}
}

\uncover<3->{
We select $\vy=-\eta\A(\jac \vf(\vx))\in\Rn$.
}
\end{frame}

\subsection{Jacobian descent}
\begin{frame}[t]{Jacobian descent}
\vspace{1cm}
The mapping $\A:\Rmn\to\Rn$ is called an {\bf aggregator}, it parameterizes {\bf Jacobian descent}:

\vspace{0.5cm}
\only<2->{
\begin{algorithm}[H]
\caption{Jacobian descent with aggregator $\A$}
\DontPrintSemicolon
\KwIn{$\vx\in\Rn$, $0<\eta$, $T\in\mathbb N$, $\A:\Rmn\to\Rn$}
\For{$t \gets 1$ \KwTo $T$}{
    $\vx\gets \vx - \eta \A(\jac \vf(\vx))$\;
}
\KwOut{$\vx$}
\end{algorithm}}
\end{frame}

\subsection{Formal definition of UPGrad}
\begin{frame}{Formal definition of UPGrad}
Given $J\in\Rmn$, let ${\color{color1}C_J}$ be the cone $\{ \vy\in\Rn : \0\pleq J\vy \}$.
$\A$ is non-conflicting if for all $J\in\Rmn$, $\A(J)\in {\color{color1}C_J}$.

For any $\vx\in\Rn$, the projection of $\vx$ onto ${\color{color1}C_J}$ is
\begin{flalign*}
\proj_J(\vx)=\argmin_{\vy\in {\color{color1}C_J}}\| \vx-\vy\|^2&&
\end{flalign*}
\begin{flalign*}
\upgrad(J)=\frac{1}{m}\sum_{i\in\range{m}} \proj_J\left(J^\T \ve_i\right)&&
\end{flalign*}

\end{frame}

  features = backbone(x)
  y_pred1 = head1(features)
  y_pred2 = head2(features)
  loss1 = loss_fn(y_pred1, y1)
  loss2 = loss_fn(y_pred2, y2)

  mtl_backward([loss1, loss2], features)
  jac_to_grad(backbone.parameters(), UPGrad())
  optimizer.step()
  optimizer.zero_grad()

\begin{frame}{Bonus: Gramian-based JD}
  \uncover<1->{$\upgrad(\jac\vf(\vx))$ is a linear combination of the rows of $\jac\vf(\vx)$.}

  \uncover<2->{The weights only depend on the (small) Gramian $\gram\vf(\vx) = \jac\vf(\vx) \cdot \jac\vf(\vx)^\T$.}

  \uncover<3->{We can compute $\gram\vf(\vx)$ directly, extract the weights, combine the losses, and do a step of gradient descent.}

  \uncover<4->{We have developed the \textbf{\texttt{autogram}} engine to compute $\gram\vf(\vx)$ efficiently.}

  \uncover<5->{This saves a lot of memory, so it's often much faster.}

  \uncover<6->{Applicable to most other aggregators from the literature.}
\end{frame}
