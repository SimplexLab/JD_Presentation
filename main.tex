\subsection{Unconflicting projection of gradients (UPGrad)}
\begin{frame}{Unconflicting projection of gradients (UPGrad)}

% Given $J\in\Rmn$\uncover<2->{, let ${\color{color1}C_J}$ be the cone $\{ \vy\in\Rn : \0\pleq J\vy \}$.
% $\A$ is non-conflicting if for all $J\in\Rmn$, $\A(J)\in {\color{color1}C_J}$.}

% \uncover<3->{For any $\vx\in\Rn$, the projection of $\vx$ onto ${\color{color1}C_J}$ is
% \begin{flalign*}
% \proj_J(\vx)=\argmin_{\vy\in {\color{color1}C_J}}\| \vx-\vy\|^2&&
% \end{flalign*}}
% \uncover<4->{
% \begin{flalign*}
% \upgrad(J)=\frac{1}{m}\sum_{i\in\range{m}} \proj_J\left(J^\T \ve_i\right)&&
% \end{flalign*}}

\begin{frame}{Formal definition of UPGrad}
Given $J\in\Rmn$, let ${\color{color1}C_J}$ be the cone $\{ \vy\in\Rn : \0\pleq J\vy \}$.
% $\A$ is non-conflicting if for all $J\in\Rmn$, $\A(J)\in {\color{color1}C_J}$.

% For any $\vx\in\Rn$, the projection of $\vx$ onto ${\color{color1}C_J}$ is
\begin{flalign*}
\textcolor{darkgray}{\proj_J(\vx)}=\argmin_{\vy\in {\color{color1}C_J}}\| \vx-\vy\|^2&&
\end{flalign*}
\begin{flalign*}
\textcolor{blue!60!black}{\upgrad(J)}=\frac{1}{m}\sum_{i\in\range{m}} \textcolor{darkgray}{\proj_J\left(\vg_i\right)}&&
\end{flalign*}

\end{frame}
